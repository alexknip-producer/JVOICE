cmake_minimum_required(VERSION 3.25)
project(JiantVoice VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_engine.py
    ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
add_library(voice_engine STATIC src/VoiceEngine.c)
target_include_directories(voice_engine PUBLIC src PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(voice_engine PRIVATE -fwrapv)
endif()
add_executable(engine_test tests/engine_test.cpp)
target_link_libraries(engine_test PRIVATE voice_engine)
if(NOT WIN32)
    target_link_libraries(engine_test PRIVATE m)
endif()
enable_testing()
add_test(NAME engine COMMAND engine_test)
option(VOICE_BUILD_PLUGIN "Build VST3 (requires Steinberg SDK)" ON)
if(VOICE_BUILD_PLUGIN)
    if(NOT VST3_SDK_ROOT)
        message(FATAL_ERROR "Set VST3_SDK_ROOT to a recursive checkout of steinbergmedia/vst3sdk")
    endif()
    set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
    add_subdirectory(${VST3_SDK_ROOT} sdk)
    if(MINGW)
        # Unused optional SDK facility requires std::aligned_alloc, absent on MinGW.
        get_target_property(VOICE_SDK_SOURCES sdk SOURCES)
        list(FILTER VOICE_SDK_SOURCES EXCLUDE REGEX "/utility/dataexchange.cpp$")
        set_property(TARGET sdk PROPERTY SOURCES ${VOICE_SDK_SOURCES})
        target_link_options(validator PRIVATE -municode -static -static-libgcc -static-libstdc++)
        target_link_options(engine_test PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_executable(plugin_test tests/plugin_test.cpp
        ${VST3_SDK_ROOT}/public.sdk/source/vst/vstsinglecomponenteffect.cpp)
    target_link_libraries(plugin_test PRIVATE voice_engine sdk sdk_hosting)
    target_compile_definitions(plugin_test PRIVATE NOMINMAX UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(plugin_test PRIVATE /utf-8 /UNDEBUG)
    else()
        target_compile_options(plugin_test PRIVATE -UNDEBUG -O0)
    endif()
    if(WIN32)
        target_link_libraries(plugin_test PRIVATE user32 gdi32)
    endif()
    if(MINGW)
        target_link_options(plugin_test PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME plugin COMMAND plugin_test)
    add_library(JiantVoice MODULE src/Plugin.cpp
        ${VST3_SDK_ROOT}/public.sdk/source/vst/vstsinglecomponenteffect.cpp)
    if(WIN32)
        target_sources(JiantVoice PRIVATE ${VST3_SDK_ROOT}/public.sdk/source/main/dllmain.cpp)
        target_link_libraries(JiantVoice PRIVATE user32 gdi32)
    else()
        target_sources(JiantVoice PRIVATE ${VST3_SDK_ROOT}/public.sdk/source/main/linuxmain.cpp)
    endif()
    target_link_libraries(JiantVoice PRIVATE voice_engine sdk)
    target_compile_definitions(JiantVoice PRIVATE NOMINMAX UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(JiantVoice PRIVATE /utf-8)
        target_compile_options(engine_test PRIVATE /utf-8 /UNDEBUG)
    else()
        target_compile_options(engine_test PRIVATE -UNDEBUG)
    endif()
    if(MINGW)
        target_link_options(JiantVoice PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    if(WIN32)
        set(VOICE_BUNDLE ${CMAKE_CURRENT_BINARY_DIR}/JVoice.vst3/Contents/x86_64-win)
        set_target_properties(JiantVoice PROPERTIES OUTPUT_NAME JVoice PREFIX "" SUFFIX ".vst3" LIBRARY_OUTPUT_DIRECTORY ${VOICE_BUNDLE}
            LIBRARY_OUTPUT_DIRECTORY_RELEASE ${VOICE_BUNDLE}
            RUNTIME_OUTPUT_DIRECTORY ${VOICE_BUNDLE} RUNTIME_OUTPUT_DIRECTORY_RELEASE ${VOICE_BUNDLE})
    else()
        set(VOICE_BUNDLE ${CMAKE_CURRENT_BINARY_DIR}/JVoice.vst3/Contents/x86_64-linux)
        set_target_properties(JiantVoice PROPERTIES OUTPUT_NAME JVoice PREFIX "" LIBRARY_OUTPUT_DIRECTORY ${VOICE_BUNDLE})
    endif()
endif()
